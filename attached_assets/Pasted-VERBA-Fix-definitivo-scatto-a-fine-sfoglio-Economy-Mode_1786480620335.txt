INTERVENTO 2 — SOSTITUISCI INTERAMENTE src/components/ParallaxPager.tsx

import { useRef, useState, useLayoutEffect, memo, type ReactNode } from "react";
import { motion, useMotionValue, useTransform, animate, type MotionValue } from "framer-motion";

/** Quanto ogni banda "resta indietro" rispetto al dito. Banda 0 = parola (segue 1:1). */
const LAG = 0.055;
/** Frazione di larghezza oltre la quale il rilascio conferma il cambio pagina. */
const COMMIT_RATIO = 0.28;
/** Velocità (px/s) oltre la quale un flick conferma comunque il cambio pagina. */
const FLICK_VELOCITY = 450;
/** Molla del carosello: morbida, senza rimbalzo percepibile. */
const SPRING = { type: "spring" as const, stiffness: 190, damping: 26, mass: 0.9, restDelta: 0.5 };

const Band = memo(function Band({
  x,
  k,
  children,
}: {
  x: MotionValue<number>;
  k: number;
  children: ReactNode;
}) {
  const bx = useTransform(x, (v) => -v * (k * LAG));
  if (k === 0) return <div>{children}</div>;
  return <motion.div style={{ x: bx, willChange: "transform" }}>{children}</motion.div>;
});

const Slide = memo(function Slide({
  x,
  offset,
  bands,
}: {
  x: MotionValue<number>;
  offset: number;
  bands: ReactNode[];
}) {
  return (
    <div
      style={
        offset === 0
          ? { position: "relative", width: "100%" }
          : { position: "absolute", top: 0, left: `${offset * 100}%`, width: "100%" }
      }
    >
      {bands.map((b, k) => (
        <Band key={k} x={x} k={k}>
          {b}
        </Band>
      ))}
    </div>
  );
});

interface ParallaxPagerProps {
  index: number;
  total: number;
  onIndexChange: (next: number) => void;
  renderBands: (i: number) => ReactNode[];
}

export default function ParallaxPager({ index, total, onIndexChange, renderBands }: ParallaxPagerProps) {
  const x = useMotionValue(0);
  const wrapRef = useRef<HTMLDivElement>(null);
  const [width, setWidth] = useState(0);

  const canPrev = index > 0;
  const canNext = index < total - 1;

  // Cache degli elementi JSX per indice. Riusare lo STESSO oggetto elemento
  // fa sì che React salti del tutto il re-render di quel sottoalbero.
  const cacheRef = useRef<Map<number, ReactNode[]>>(new Map());
  const lastRenderBands = useRef(renderBands);
  if (lastRenderBands.current !== renderBands) {
    cacheRef.current.clear();
    lastRenderBands.current = renderBands;
  }
  const bandsFor = (i: number): ReactNode[] => {
    const cache = cacheRef.current;
    if (!cache.has(i)) {
      cache.set(i, renderBands(i));
      if (cache.size > 7) {
        for (const key of Array.from(cache.keys())) {
          if (Math.abs(key - i) > 2) cache.delete(key);
        }
      }
    }
    return cache.get(i) as ReactNode[];
  };

  useLayoutEffect(() => {
    const el = wrapRef.current;
    if (!el) return;
    const measure = () => setWidth(el.offsetWidth);
    measure();
    const ro = new ResizeObserver(measure);
    ro.observe(el);
    return () => ro.disconnect();
  }, []);

  const W = width || 1;

  const handleDragEnd = (
    _e: unknown,
    info: { offset: { x: number }; velocity: { x: number } }
  ) => {
    const dist = info.offset.x;
    const vel = info.velocity.x;

    let step = 0;
    if ((dist < -W * COMMIT_RATIO || vel < -FLICK_VELOCITY) && canNext) step = 1;
    else if ((dist > W * COMMIT_RATIO || vel > FLICK_VELOCITY) && canPrev) step = -1;

    if (step !== 0) {
      // Commit SUBITO, non a fine molla: si compensa x di una larghezza schermo
      // così visivamente non cambia nulla, ma il lavoro di React avviene ora
      // (mascherato dal rilascio del dito) invece che sull'ultimo frame.
      x.jump(x.get() + step * W);
      onIndexChange(index + step);
    }

    animate(x, 0, { ...SPRING, velocity: vel });
  };

  return (
    <div
      ref={wrapRef}
      style={{
        position: "relative",
        width: "100%",
        clipPath: "inset(-2000px 0 -2000px 0)",
        touchAction: "pan-y",
      }}
    >
      <motion.div
        drag={total > 1 ? "x" : false}
        dragConstraints={{ left: canNext ? -W : 0, right: canPrev ? W : 0 }}
        dragElastic={0.22}
        dragMomentum={false}
        onDragEnd={handleDragEnd}
        style={{
          x,
          position: "relative",
          width: "100%",
          cursor: "grab",
          willChange: "transform",
          backfaceVisibility: "hidden",
        }}
      >
        {canPrev && <Slide key={index - 1} x={x} offset={-1} bands={bandsFor(index - 1)} />}
        <Slide key={index} x={x} offset={0} bands={bandsFor(index)} />
        {canNext && <Slide key={index + 1} x={x} offset={1} bands={bandsFor(index + 1)} />}
      </motion.div>
    </div>
  );
}
