import { useEffect, useMemo, useRef, useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import { SCREEN_MAX } from "@/components/ScreenColumn";
import { tapScale } from "@/components/SpringTap";
import type { QuestionProps } from "@/components/quiz/types";

// Gradino 2 — Recall in context.
//
// Lo sfondo non decora: è la materia della risposta. Quando rispondi, il
// bottone giusto si svuota per intero — le sue lettere salgono nel vortice,
// il vortice accelera girando e scaglia fuori quelle che non servono, poi
// rallenta e si posa nella frase. Sull'errore lo stesso movimento si compie
// con le lettere della risposta PREMUTA, ma il vortice perde coesione, si
// allarga e ognuna rientra da dove è venuta: il bottone si ricompone e poi
// parte il giro della parola giusta.
//
// Tutto è governato da un solo orologio (animRef.current.t0): le lettere non
// hanno animazioni proprie, sono la conseguenza di dove si trova la sequenza.

/* ── numeri regolabili a occhio, senza agent ── */
const DUR = 1600;        // durata di un giro completo, ms
const TURNS = 1.9;       // giri del vortice
const RAD = 64;          // raggio dell'anello, px
const SPIN = 1.3;        // rotazione propria di ogni lettera
const SWEEP = 4;         // forza con cui le lettere in eccesso vengono scagliate
const PA = 0.30;         // fine della salita
const PB = 0.72;         // fine del vortice, inizio della posa
const ORBIT_Y = 0.12;    // quota del centro dentro l'altezza del buco
const ORBIT_RY = 9;      // ampiezza verticale dell'orbita a riposo
const GAP_MIN = 52;      // larghezza fissa: un buco largo quanto la risposta la regala
const ORBIT_N = 6;
const DENSITY_PX2 = 9000;
const FREQ = "eeeeeetttttaaaaooooiiiinnnnsssrrrhhllddccuummffggppywwbvkxjqz";

/* ritardi di rivelazione, ricavati dalla durata: la scheda non deve salire
   mentre la parola si sta ancora componendo */
export const REVEAL_OK = Math.round(DUR * 1.06) + 120;
export const REVEAL_KO = Math.round(DUR * 1.06) + 260 + Math.round(DUR * 1.06) + 120;

type Role = "field" | "orbit" | "seq" | "swept" | "landed" | "held";
type Target = { ch: string; x: number; y: number; size: number };
type Letter = {
  ch: string; x: number; y: number; vx: number; vy: number;
  size: number; lav: boolean; base: number; a: number; rot: number;
  role: Role; phase: number; spin: number; wob: number;
  sx: number; sy: number; a0: number; jit: number;
  tgt: Target | null; ox: number; oy: number;
  origRole: Role | null; dom: HTMLElement | null;
  sweepAt: number; swept: boolean; dead: boolean;
};
type Anim = { t0: number; ok: boolean; chosen: Letter[] };

const rnd = (a: number, b: number) => a + Math.random() * (b - a);
const easeOut = (p: number) => 1 - Math.pow(1 - p, 3);
const easeIn = (p: number) => p * p;
const easeInOut = (p: number) => (p < 0.5 ? 2 * p * p : 1 - Math.pow(-2 * p + 2, 2) / 2);
const clamp01 = (p: number) => Math.max(0, Math.min(1, p));

function shuffleArray<T>(arr: T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function mkLetter(o: Partial<Letter>): Letter {
  return {
    ch: "e", x: 0, y: 0, vx: 0, vy: 0, size: 13, lav: false, base: 0.1, a: 0, rot: 0,
    role: "field", phase: 0, spin: 1, wob: rnd(20, 25), sx: 0, sy: 0, a0: 0, jit: 0,
    tgt: null, ox: 0, oy: 0, origRole: null, dom: null,
    sweepAt: 0, swept: false, dead: false, ...o,
  };
}

export default function ContextQuestion({
  word, isAnswered, selectedOption, onSelect, animKey,
}: QuestionProps) {
  const hostRef = useRef<HTMLDivElement | null>(null);
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const gapRef = useRef<HTMLSpanElement | null>(null);
  const sizerRef = useRef<HTMLSpanElement | null>(null);
  const glyphsRef = useRef<HTMLSpanElement | null>(null);
  const optsRef = useRef<HTMLDivElement | null>(null);

  const lettersRef = useRef<Letter[]>([]);
  const animRef = useRef<Anim | null>(null);
  const rafRef = useRef<number | null>(null);
  const sizeRef = useRef({ w: 0, h: 0 });
  const gcRef = useRef({ x: 0, y: 0 });
  const slowRef = useRef(0.55);

  const [reduced] = useState(
    () => typeof window !== "undefined" &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches,
  );

  const answer = word.word;
  const parts = useMemo(() => {
    const s = word.contextStem ?? "";
    const i = s.indexOf("____");
    if (i < 0) return { pre: s, post: "" };
    return { pre: s.slice(0, i), post: s.slice(i + 4) };
  }, [word.contextStem]);

  const options = useMemo(
    () => shuffleArray([answer, ...(word.contextDistractors ?? [])]),
    [word.id],
  );

  /* ── campo e vortice ─────────────────────────────────────────────────── */
  function buildField() {
    const { w, h } = sizeRef.current;
    if (!w || !h) return;
    const out: Letter[] = [];
    const n = Math.max(8, Math.round((w * h) / DENSITY_PX2));
    for (let i = 0; i < n; i++) {
      out.push(mkLetter({
        ch: FREQ[(Math.random() * FREQ.length) | 0], x: rnd(0, w), y: rnd(0, h),
        vx: rnd(-0.34, 0.34), vy: rnd(-0.24, 0.24), size: rnd(9, 17),
        lav: Math.random() < 0.3, base: rnd(0.06, 0.12), role: "field",
        spin: rnd(0.5, 1.1), phase: Math.random() * 6.283,
      }));
    }
    // Il vortice: lettere CASUALI, ripescate a ogni parola. Non sono la
    // risposta — sarebbe un indizio regalato — ma sono materia disponibile.
    for (let i = 0; i < ORBIT_N; i++) {
      out.push(mkLetter({
        ch: FREQ[(Math.random() * FREQ.length) | 0], size: rnd(12, 16),
        lav: i % 2 === 0, base: 0.4, role: "orbit",
        spin: rnd(0.6, 0.9), phase: (i / ORBIT_N) * 6.283,
      }));
    }
    lettersRef.current = out;
  }

  function measureGap() {
    const host = hostRef.current, gap = gapRef.current;
    if (!host || !gap) return;
    const s = host.getBoundingClientRect(), r = gap.getBoundingClientRect();
    gcRef.current = {
      x: r.left - s.left + r.width / 2,
      y: r.top - s.top + r.height * ORBIT_Y,
    };
  }

  function targetsFor(w: string, open: boolean): Target[] {
    const host = hostRef.current, gap = gapRef.current;
    const sizer = sizerRef.current, glyphs = glyphsRef.current;
    if (!host || !gap || !sizer || !glyphs) return [];
    sizer.textContent = w;
    glyphs.innerHTML = w.split("").map((c) => `<span>${c}</span>`).join("");
    const prev = gap.style.transition;
    gap.style.transition = "none";
    gap.style.width = "auto";
    void gap.offsetHeight;
    const full = Math.ceil(gap.getBoundingClientRect().width);
    const s = host.getBoundingClientRect();
    const out = [...glyphs.children].map((g, i) => {
      const r = (g as HTMLElement).getBoundingClientRect();
      return {
        ch: w[i], x: r.left - s.left + r.width / 2, y: r.top - s.top + r.height / 2,
        size: parseFloat(getComputedStyle(g as HTMLElement).fontSize),
      };
    });
    gap.style.width = GAP_MIN + "px";
    void gap.offsetHeight;
    gap.style.transition = prev || "";
    if (open) requestAnimationFrame(() => {
      if (gapRef.current) gapRef.current.style.width = full + "px";
    });
    return out;
  }

  /** Svuota un bottone: OGNI sua lettera diventa materia. Niente parole bucate. */
  function drain(optIdx: number): Letter[] {
    const wrap = optsRef.current, host = hostRef.current;
    if (!wrap || !host) return [];
    const el = wrap.children[optIdx] as HTMLElement | undefined;
    if (!el || el.dataset.drained) return [];
    el.dataset.drained = "1";
    const s = host.getBoundingClientRect();
    const out: Letter[] = [];
    [...el.querySelectorAll("span")].forEach((g) => {
      const node = g as HTMLElement;
      const r = node.getBoundingClientRect();
      node.style.opacity = "0";
      const L = mkLetter({
        ch: (node.textContent || "").toLowerCase(),
        x: r.left - s.left + r.width / 2, y: r.top - s.top + r.height / 2,
        size: parseFloat(getComputedStyle(node).fontSize),
        lav: true, base: rnd(0.07, 0.13), role: "held", dom: node,
      });
      L.ox = L.x; L.oy = L.y;
      lettersRef.current.push(L);
      out.push(L);
    });
    return out;
  }

  /** Avvia l'unico orologio. */
  function start(w: string, btnIdx: number, ok: boolean) {
    const pool = drain(btnIdx);
    const chosen: Letter[] = [];
    const gc = gcRef.current;
    let tgs: Target[] | null = null;

    if (ok) {
      tgs = targetsFor(w, true);
      const used = new Set<Letter>();
      for (const tg of tgs) {
        let best: Letter | null = null, bd = Infinity;
        const scan = (arr: Letter[], k: number) => {
          for (const L of arr) {
            if (used.has(L) || L.ch !== tg.ch) continue;
            const d = Math.hypot(L.x - gc.x, L.y - gc.y) * k;
            if (d < bd) { bd = d; best = L; }
          }
        };
        scan(pool, 0.3);
        if (!best) scan(lettersRef.current.filter((L) => L.role === "orbit"), 0.5);
        if (!best) scan(lettersRef.current.filter((L) => L.role === "field"), 1);
        if (!best) {
          const { w: cw, h: ch } = sizeRef.current;
          const L = mkLetter({
            ch: tg.ch, x: tg.x < cw / 2 ? -24 : cw + 24, y: rnd(ch * 0.25, ch * 0.7),
            size: 13, base: 0.09, role: "field",
          });
          L.ox = L.x; L.oy = L.y;
          lettersRef.current.push(L);
          best = L;
        }
        used.add(best);
        chosen.push(best);
      }
    } else {
      // a vorticare è la risposta PREMUTA, insieme al vortice
      chosen.push(...pool, ...lettersRef.current.filter((L) => L.role === "orbit"));
    }

    const inUse = new Set(chosen);
    const swept = ok
      ? lettersRef.current.filter(
          (L) => (L.role === "orbit" || pool.includes(L)) && !inUse.has(L))
      : [];

    chosen.forEach((L, i) => {
      if (L.origRole === null) L.origRole = L.dom ? null : L.role;
      if (!L.ox && !L.oy) { L.ox = L.x; L.oy = L.y; }
      L.role = "seq"; L.sx = L.x; L.sy = L.y;
      L.a0 = (i / chosen.length) * 6.283;
      L.tgt = tgs ? tgs[i] : null;
      L.jit = rnd(-1, 1);
    });
    swept.forEach((L, i) => {
      L.role = "swept"; L.sweepAt = 0.16 + i * 0.055; L.swept = false;
    });
    animRef.current = { t0: performance.now(), ok, chosen };
  }

  /* ── il disegno ──────────────────────────────────────────────────────── */
  function draw(now: number) {
    const cv = canvasRef.current, host = hostRef.current;
    if (!cv || !host) return;
    const ctx = cv.getContext("2d");
    if (!ctx) return;
    measureGap();

    if (lettersRef.current.some((L) => L.dead))
      lettersRef.current = lettersRef.current.filter((L) => !L.dead);

    const { w, h } = sizeRef.current;
    const dpr = Math.min(window.devicePixelRatio || 1, 3);
    const t = now / 1000;
    const gc = gcRef.current;
    const anim = animRef.current;

    let P = 0;
    if (anim) {
      P = (now - anim.t0) / DUR;
      if (P >= 1.06) {
        if (!anim.ok) {
          anim.chosen.forEach((L) => {
            if (L.dom) { L.dom.style.opacity = "1"; L.dead = true; }
            else {
              L.role = L.origRole || "field"; L.a = L.base;
              L.x = L.ox; L.y = L.oy; L.vx = rnd(-0.3, 0.3); L.vy = rnd(-0.2, 0.2);
            }
          });
          const wrap = optsRef.current;
          if (wrap) [...wrap.children].forEach((el) => delete (el as HTMLElement).dataset.drained);
        }
        const wasOk = anim.ok;
        animRef.current = null;
        if (!wasOk) window.setTimeout(() => {
          start(answer, options.indexOf(answer), true);
        }, 260);
      }
    }

    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, w, h);
    ctx.textAlign = "center";
    ctx.textBaseline = "middle";

    for (const L of lettersRef.current) {
      if (L.role === "seq" && animRef.current) {
        const a = animRef.current;
        const p = clamp01(P);
        const spinP = clamp01((p - PA * 0.55) / (PB - PA * 0.55));
        const ang = L.a0 + (easeIn(spinP) * 0.55 + spinP * 0.45) * TURNS * 6.283;
        const rr = RAD * (1 - 0.42 * easeInOut(spinP));
        const ox = gc.x + Math.cos(ang) * rr;
        const oy = gc.y + Math.sin(ang) * rr * 0.42;

        if (p < PA) {                                  // salita, con arco
          const q = easeOut(p / PA);
          const mx = (L.sx + ox) / 2, my = Math.min(L.sy, oy) - 70;
          const u = 1 - q;
          L.x = u * u * L.sx + 2 * u * q * mx + q * q * ox;
          L.y = u * u * L.sy + 2 * u * q * my + q * q * oy;
          L.a = 0.25 + 0.75 * q; L.rot = q * SPIN * 2;
        } else if (p < PB) {                           // il vortice gira
          const wob = a.ok ? 0 : ((p - PA) / (PB - PA)) * L.jit * 9;
          L.x = ox + wob; L.y = oy + wob * 0.4; L.a = 1;
          L.rot = ang * SPIN;
        } else {                                       // si posa — oppure si sfalda
          const q = clamp01((p - PB) / (1 - PB)), e = easeOut(q);
          if (a.ok && L.tgt) {
            L.x = ox + (L.tgt.x - ox) * e;
            L.y = oy + (L.tgt.y - oy) * e;
            L.rot = ang * SPIN * (1 - e);
            L.size = L.size + (L.tgt.size - L.size) * 0.12;
            L.a = 1;
            if (q >= 1) { L.role = "landed"; L.x = L.tgt.x; L.y = L.tgt.y; L.rot = 0; }
          } else {
            const flare = 1 + e * 1.5;
            const fx = gc.x + Math.cos(ang) * rr * flare;
            const fy = gc.y + Math.sin(ang) * rr * 0.42 * flare;
            const back = easeInOut(Math.max(0, (q - 0.28) / 0.72));
            L.x = fx + (L.ox - fx) * back;
            L.y = fy + (L.oy - fy) * back;
            L.rot = ang * SPIN * (1 - back);
            L.a = Math.max(0.12, 1 - back * 0.85);
          }
        }
      } else if (L.role === "swept" && animRef.current) {
        if (!L.swept && P >= L.sweepAt) {
          L.swept = true;
          const a = Math.atan2(L.y - gc.y, L.x - gc.x) + rnd(-0.4, 0.4);
          L.vx = Math.cos(a) * SWEEP; L.vy = Math.sin(a) * SWEEP * 0.7;
        }
        if (L.swept) {
          L.x += L.vx; L.y += L.vy; L.vx *= 0.965; L.vy *= 0.965;
          L.rot += 0.06; L.a = Math.max(0, L.a - 0.012);
          if (L.a <= 0.02) {
            L.role = "field"; L.base = rnd(0.05, 0.1); L.a = L.base; L.rot = 0;
            L.x = rnd(0, w); L.y = rnd(0, h); L.vx = rnd(-0.3, 0.3); L.vy = rnd(-0.2, 0.2);
          }
        } else {
          const a = t * L.spin + L.phase;
          L.x = gc.x + Math.cos(a) * L.wob; L.y = gc.y + Math.sin(a) * ORBIT_RY;
          L.a = L.base;
        }
      } else if (L.role === "landed") { L.a = 1; }
      else if (L.role === "held") { L.a = 0; }
      else if (L.role === "orbit") {
        const a = t * L.spin + L.phase;
        L.x = gc.x + Math.cos(a) * L.wob + Math.cos(a * 2.3) * 3;
        L.y = gc.y + Math.sin(a) * ORBIT_RY + Math.sin(a * 1.7) * 1.8;
        L.a = L.base * (0.5 + 0.5 * Math.abs(Math.sin(a * 0.8)));
        L.rot = 0;
      } else {
        const s = slowRef.current;
        L.x += L.vx * s; L.y += L.vy * s; L.vx *= 0.982; L.vy *= 0.982;
        if (L.x < -16) L.x = w + 12; if (L.x > w + 16) L.x = -12;
        if (L.y < -16) L.y = h + 12; if (L.y > h + 16) L.y = -12;
        L.a = L.base; L.rot = 0;
      }

      if (L.role === "held") continue;
      const seq = animRef.current;
      const col = L.role === "landed" || (L.role === "seq" && seq && seq.ok && P >= PB)
        ? "52,211,153"
        : L.role === "seq" && seq && !seq.ok && P >= PB
          ? "239,68,68"
          : L.lav ? "199,184,232" : "245,158,11";
      ctx.save();
      ctx.translate(L.x, L.y);
      if (L.rot) ctx.rotate(L.rot);
      ctx.font = `${L.role === "seq" || L.role === "landed" ? 500 : 400} ${L.size.toFixed(1)}px 'Space Grotesk', sans-serif`;
      ctx.fillStyle = `rgba(${col},${clamp01(L.a).toFixed(3)})`;
      ctx.fillText(L.ch, 0, 0);
      ctx.restore();
    }
  }

  /* ── ciclo di vita ───────────────────────────────────────────────────── */
  useEffect(() => {
    if (reduced) return;
    const host = hostRef.current, cv = canvasRef.current;
    if (!host || !cv) return;
    const resize = () => {
      const r = host.getBoundingClientRect();
      sizeRef.current = { w: r.width, h: r.height };
      const dpr = Math.min(window.devicePixelRatio || 1, 3);
      cv.width = Math.round(r.width * dpr);
      cv.height = Math.round(r.height * dpr);
      buildField();
      measureGap();
    };
    resize();
    window.addEventListener("resize", resize);
    const loop = (now: number) => { draw(now); rafRef.current = requestAnimationFrame(loop); };
    rafRef.current = requestAnimationFrame(loop);
    return () => {
      window.removeEventListener("resize", resize);
      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
    };
  }, [reduced]);

  // nuova parola: campo rifatto, buco richiuso, orologio azzerato
  useEffect(() => {
    animRef.current = null;
    slowRef.current = 0.55;
    if (gapRef.current) gapRef.current.style.width = GAP_MIN + "px";
    if (glyphsRef.current) glyphsRef.current.innerHTML = "";
    if (sizerRef.current) sizerRef.current.textContent = "";
    const wrap = optsRef.current;
    if (wrap) [...wrap.children].forEach((el) => {
      const e = el as HTMLElement;
      delete e.dataset.drained;
      [...e.querySelectorAll("span")].forEach((s) => ((s as HTMLElement).style.opacity = "1"));
    });
    if (!reduced) { buildField(); measureGap(); }
  }, [word.id, reduced]);

  function handlePick(option: string, i: number) {
    if (isAnswered) return;
    const correct = option === answer;
    slowRef.current = 1;
    onSelect(option, correct, correct ? REVEAL_OK : REVEAL_KO);
    if (reduced) {
      if (gapRef.current) gapRef.current.style.width = "auto";
      if (sizerRef.current) {
        sizerRef.current.textContent = answer;
        sizerRef.current.style.visibility = "visible";
      }
      return;
    }
    measureGap();
    start(correct ? answer : option, i, correct);
  }

  const optStyle = (o: string): React.CSSProperties => {
    if (!isAnswered)
      return { background: "rgba(10,10,10,0.5)", border: "1px solid rgba(199,184,232,0.16)", color: "#C7B8E8" };
    if (o === answer)
      return { background: "rgba(52,211,153,0.05)", border: "1px solid rgba(52,211,153,0.55)", color: "#34D399" };
    if (o === selectedOption)
      return { background: "rgba(239,68,68,0.06)", border: "1px solid rgba(239,68,68,0.6)", color: "#FCA5A5" };
    return { background: "rgba(10,10,10,0.5)", border: "1px solid rgba(199,184,232,0.16)", color: "#C7B8E8", opacity: 0.14 };
  };

  return (
    <div ref={hostRef}
      style={{ position: "relative", width: "100%", maxWidth: SCREEN_MAX, flex: 1, display: "flex", flexDirection: "column" }}>
      <canvas ref={canvasRef} aria-hidden
        style={{ position: "absolute", inset: 0, width: "100%", height: "100%", pointerEvents: "none", zIndex: 0 }} />

      <AnimatePresence mode="wait">
        <motion.div key={animKey}
          initial={{ opacity: 0, y: 16 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -16 }}
          transition={{ duration: 0.3, ease: "easeOut" }}
          style={{ position: "relative", zIndex: 1, paddingTop: 28, paddingBottom: 22 }}>
          <p style={{ fontFamily: "'Inter', sans-serif", fontSize: "0.66rem", letterSpacing: "0.14em", textTransform: "uppercase", color: "rgba(255,255,255,0.32)", textAlign: "center", margin: "0 0 18px" }}>
            Complete the sentence
          </p>
          <p style={{ fontFamily: "'Space Grotesk', sans-serif", fontWeight: 400, fontSize: "clamp(17px, 4.6vw, 21px)", lineHeight: 1.72, color: "rgba(255,255,255,0.93)", textAlign: "center", margin: 0 }}>
            {parts.pre}
            <span ref={gapRef}
              style={{ position: "relative", display: "inline-block", verticalAlign: "baseline", width: GAP_MIN, transition: "width 0.4s cubic-bezier(.19,1,.22,1)" }}>
              <span ref={sizerRef} style={{ visibility: "hidden", whiteSpace: "nowrap", fontFamily: "'Space Grotesk', sans-serif", fontWeight: 500 }} />
              <span ref={glyphsRef} aria-hidden style={{ position: "absolute", left: 0, top: 0, whiteSpace: "nowrap", visibility: "hidden", fontFamily: "'Space Grotesk', sans-serif", fontWeight: 500 }} />
            </span>
            {parts.post}
          </p>
        </motion.div>
      </AnimatePresence>

      <div ref={optsRef}
        style={{ position: "relative", zIndex: 1, display: "flex", flexDirection: "column", gap: 9, marginTop: "auto" }}>
        {options.map((o, i) => (
          <motion.button
            key={`${word.id}-${o}`}
            data-testid={`context-option-${i}`}
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ delay: i * 0.06, duration: 0.25, ease: "easeOut" }}
            whileTap={isAnswered ? undefined : tapScale("card")}
            onClick={() => handlePick(o, i)}
            disabled={isAnswered}
            style={{
              ...optStyle(o), borderRadius: 12, padding: "13px 16px",
              cursor: isAnswered ? "default" : "pointer",
              fontFamily: "'Space Grotesk', sans-serif", fontWeight: 500,
              fontSize: "1rem", textAlign: "center", outline: "none",
              transition: "border-color 0.3s, background 0.3s, color 0.3s, opacity 0.45s",
            }}>
            {/* ogni lettera è un nodo suo: è così che può staccarsi e volare */}
            {o.split("").map((c, k) => (
              <span key={k} style={{ display: "inline-block", transition: "opacity 0.1s linear" }}>{c}</span>
            ))}
          </motion.button>
        ))}
      </div>
    </div>
  );
}
